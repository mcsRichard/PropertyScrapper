# 腾讯云SCF环境变量配置模板
# 在云函数控制台的"环境变量"中配置以下变量

# ============================================
# 数据库配置（TDSQL-C Serverless）
# ============================================
DB_HOST=你的数据库地址
DB_PORT=29493
DB_USER=你的数据库用户名
DB_PASSWORD=你的数据库密码
DB_NAME=ukestate

# ============================================
# 腾讯云COS配置（图片存储）
# ============================================
COS_SECRET_ID=你的COS密钥ID
COS_SECRET_KEY=你的COS密钥
COS_REGION=ap-shanghai
COS_BUCKET=你的存储桶名称
COS_DOMAIN=你的CDN域名（可选，如：https://cdn.yourdomain.com）

# ============================================
# AI搜索功能配置
# ============================================
# AI API类型：deepseek 或 openai
AI_API_TYPE=deepseek

# DeepSeek API配置（推荐，国内更稳定）
DEEPSEEK_API_KEY=你的DeepSeek API密钥
DEEPSEEK_API_BASE=https://api.deepseek.com
DEEPSEEK_MODEL=deepseek-chat

# OpenAI API配置（可选，如果使用OpenAI）
OPENAI_API_KEY=你的OpenAI API密钥（可选）
OPENAI_API_BASE=https://api.openai.com/v1
OPENAI_MODEL=gpt-4o-mini

# ============================================
# 注意事项
# ============================================
# 1. 所有值都不要加引号
# 2. 不要包含空格（除非值本身需要）
# 3. 敏感信息（密码、密钥）必须通过环境变量配置
# 4. 不要上传 .env 文件到云函数
# 5. 配置后需要重新部署函数才能生效
